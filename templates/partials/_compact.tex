((# ===================================================================
    Compact overrides for "fit to one page".

    Re-applies tighter geometry, section spacing, list spacing, and line
    height AFTER the template's own preamble settings, so the last definition
    wins. Driven by the integer `density` level the renderer passes:
      0 = the template's native look (this file emits nothing)
      1 = slightly tighter
      2 = tighter + reduced line height
      3 = tightest readable floor
    Each template includes this once, just before \begin{document}.
   =================================================================== #))
((* if density|default(0) >= 1 *))
\geometry{top=0.45in, bottom=0.45in, left=0.5in, right=0.5in}
\titlespacing*{\section}{0pt}{6pt}{3pt}
\setlist[itemize]{itemsep=1pt, topsep=1.5pt, parsep=0pt}
\linespread{0.99}
((* endif *))
((* if density|default(0) >= 2 *))
\geometry{top=0.4in, bottom=0.4in, left=0.45in, right=0.45in}
\titlespacing*{\section}{0pt}{4pt}{2pt}
\setlist[itemize]{itemsep=0.6pt, topsep=1pt, parsep=0pt}
\linespread{0.97}
((# Font is the strongest lever for fitting a page; drop to ~9.5pt. #))
\AtBeginDocument{\fontsize{9.5pt}{11.2pt}\selectfont}
((* endif *))
((* if density|default(0) >= 3 *))
\geometry{top=0.35in, bottom=0.35in, left=0.4in, right=0.4in}
\titlespacing*{\section}{0pt}{3pt}{1.5pt}
\linespread{0.95}
((# Readable floor: ~9pt with tight margins. Never compact past this. #))
\AtBeginDocument{\fontsize{9pt}{10.6pt}\selectfont}
((* endif *))
